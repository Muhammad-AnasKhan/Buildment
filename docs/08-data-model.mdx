## Prisma schema

```prisma
enum Role            { MENTEE MENTOR ADMIN }
enum CourseStatus    { DRAFT PUBLISHED ARCHIVED }
enum EnrollStatus    { ASSIGNED IN_PROGRESS COMPLETED DROPPED }
enum ProgressStatus  { NOT_STARTED IN_PROGRESS COMPLETED }
enum BlockType       { QUIZ TEST MUST_READ OPEN_QUESTION CODE }
enum ResponseStatus  { DRAFT SUBMITTED AUTO_GRADED PENDING_REVIEW REVIEWED NEEDS_REVISION }
enum ReviewVerdict   { APPROVED NEEDS_REVISION }
enum StudySessionStatus { RUNNING PAUSED ENDED }

model User {
  id            String   @id @default(cuid())
  email         String   @unique
  name          String?
  image         String?
  role          Role     @default(MENTEE)
  createdAt     DateTime @default(now())
  coursesOwned  Course[]     @relation("CourseMentor")
  enrollments   Enrollment[]
  responses     Response[]
  reviewsGiven  Review[]     @relation("Reviewer")
  invitesSent   Invite[]     @relation("InviteSender")
}

model Course {
  id            String       @id @default(cuid())
  slug          String       @unique
  title         String
  description   String?
  coverUrl      String?
  projectGoal   String?      // "Build a multi-vendor marketplace"
  difficulty    String?
  estimatedHours Int?
  status        CourseStatus @default(DRAFT)
  sequential    Boolean      @default(true)
  mentorId      String
  mentor        User         @relation("CourseMentor", fields: [mentorId], references: [id])
  modules       Module[]
  chapters      Chapter[]
  enrollments   Enrollment[]
  invites       Invite[]
  publishedAt   DateTime?
  createdAt     DateTime     @default(now())
  updatedAt     DateTime     @updatedAt
  @@index([mentorId, status])
}

model Module {
  id        String    @id @default(cuid())
  courseId  String
  course    Course    @relation(fields: [courseId], references: [id], onDelete: Cascade)
  title     String
  order     Int
  chapters  Chapter[]
  @@unique([courseId, order])
}

model Chapter {
  id          String   @id @default(cuid())
  courseId    String
  course      Course   @relation(fields: [courseId], references: [id], onDelete: Cascade)
  moduleId    String?
  module      Module?  @relation(fields: [moduleId], references: [id], onDelete: SetNull)
  slug        String
  title       String
  summary     String?
  order             Int
  estimatedMinutes  Int?
  readerMode        ChapterReaderMode @default(DEFAULT)
  source            String   @db.Text        // raw MDX (draft)
  compiled          String?  @db.Text        // compiled MDX at publish
  publishedAt       DateTime?
  isMilestone       Boolean  @default(false)
  blocks      Block[]
  progress    ChapterProgress[]
  @@unique([courseId, slug])
  @@index([courseId, order])
}

model Block {
  id          String    @id            // ULID authored into the MDX
  chapterId   String
  chapter     Chapter   @relation(fields: [chapterId], references: [id], onDelete: Cascade)
  type        BlockType
  order       Int
  version     Int       @default(1)
  required    Boolean   @default(true)
  blocking    Boolean   @default(false)
  points      Int       @default(1)
  config      Json                     // answers, rubric, tests — SERVER ONLY
  sourceHash  String?                  // hash of the block's MDX subtree; change detection bumps version
  tags        String[]                 // for weak-area reporting
  archivedAt  DateTime?
  responses   Response[]
  stats       BlockStats?
  @@index([chapterId, order])
}

model Enrollment {
  id                String       @id @default(cuid())
  courseId          String
  course            Course       @relation(fields: [courseId], references: [id], onDelete: Cascade)
  userId            String
  user              User         @relation(fields: [userId], references: [id], onDelete: Cascade)
  status            EnrollStatus @default(ASSIGNED)
  assignedById      String?
  percentComplete   Int          @default(0)
  chaptersCompleted Int          @default(0)
  totalScore        Int          @default(0)
  maxScore          Int          @default(0)
  pendingReviews    Int          @default(0) // denormalized for reports — maintained at Response write/review time
  lastActiveAt      DateTime?
  lastNudgeSentAt   DateTime?
  startedAt         DateTime?
  completedAt       DateTime?
  createdAt         DateTime     @default(now())
  chapterProgress   ChapterProgress[]
  dailyActivity     DailyActivity[]
  studySessions     StudySession[]
  @@unique([courseId, userId])
  @@index([userId, status])
}

model DailyActivity {
  id            String     @id @default(cuid())
  enrollmentId  String
  enrollment    Enrollment @relation(fields: [enrollmentId], references: [id], onDelete: Cascade)
  date          DateTime   @db.Date
  activeSeconds Int        @default(0)
  @@unique([enrollmentId, date])
  @@index([enrollmentId, date])
}

/// Explicit study sitting. Heartbeats credit ChapterProgress and DailyActivity only while RUNNING.
model StudySession {
  id            String             @id @default(cuid())
  enrollmentId  String
  enrollment    Enrollment         @relation(fields: [enrollmentId], references: [id], onDelete: Cascade)
  chapterId     String
  chapter       Chapter            @relation(fields: [chapterId], references: [id], onDelete: Cascade)
  status        StudySessionStatus @default(RUNNING)
  startedAt     DateTime           @default(now())
  endedAt       DateTime?
  activeSeconds Int                @default(0)
  updatedAt     DateTime           @updatedAt
  @@index([enrollmentId, startedAt])
  @@index([enrollmentId, status])
  @@index([chapterId])
}

model ChapterProgress {
  id              String         @id @default(cuid())
  enrollmentId    String
  enrollment      Enrollment     @relation(fields: [enrollmentId], references: [id], onDelete: Cascade)
  chapterId       String
  chapter         Chapter        @relation(fields: [chapterId], references: [id], onDelete: Cascade)
  status          ProgressStatus @default(NOT_STARTED)
  blocksCompleted Int            @default(0)
  blocksTotal     Int            @default(0)
  score           Int            @default(0)
  maxScore        Int            @default(0)
  timeSpentSeconds Int           @default(0)
  startedAt       DateTime?
  completedAt     DateTime?
  learningLog     Json?                    // chapter gate reflections — { version, answers }
  @@unique([enrollmentId, chapterId])
  @@index([chapterId, status])
}

model Response {
  id           String         @id @default(cuid())
  blockId      String
  block        Block          @relation(fields: [blockId], references: [id], onDelete: Cascade)
  userId       String
  user         User           @relation(fields: [userId], references: [id], onDelete: Cascade)
  enrollmentId String
  attempt      Int            @default(1)
  status       ResponseStatus @default(SUBMITTED)
  payload      Json           // shape depends on block type
  score        Int?
  maxScore     Int?
  isCorrect    Boolean?
  durationMs   Int?
  startedAt    DateTime?
  submittedAt  DateTime       @default(now())
  review       Review?
  @@index([blockId, userId, attempt])
  @@index([enrollmentId])
  @@index([status])            // drives the review queue
}

model Review {
  id         String        @id @default(cuid())
  responseId String        @unique
  response   Response      @relation(fields: [responseId], references: [id], onDelete: Cascade)
  reviewerId String
  reviewer   User          @relation("Reviewer", fields: [reviewerId], references: [id])
  score      Int?
  feedback   String        @db.Text
  verdict    ReviewVerdict
  createdAt  DateTime      @default(now())
}

enum HelpThreadStatus {
  OPEN
  RESOLVED
}

model HelpThread {
  id         String           @id @default(cuid())
  courseId   String
  course     Course           @relation(fields: [courseId], references: [id], onDelete: Cascade)
  chapterId  String?
  chapter    Chapter?         @relation(fields: [chapterId], references: [id], onDelete: SetNull)
  menteeId   String
  mentee     User             @relation("HelpThreadsAsked", fields: [menteeId], references: [id], onDelete: Cascade)
  status     HelpThreadStatus @default(OPEN)
  createdAt  DateTime         @default(now())
  updatedAt  DateTime         @updatedAt
  resolvedAt DateTime?
  messages   HelpMessage[]

  @@index([courseId, status, updatedAt])
  @@index([menteeId, updatedAt])
  @@index([chapterId, menteeId])
}

model HelpMessage {
  id        String     @id @default(cuid())
  threadId  String
  thread    HelpThread @relation(fields: [threadId], references: [id], onDelete: Cascade)
  authorId  String
  author    User       @relation("HelpMessagesAuthored", fields: [authorId], references: [id], onDelete: Cascade)
  body      String     @db.Text
  createdAt DateTime   @default(now())

  @@index([threadId, createdAt])
}

// Mentor -> mentee assignment surface (phases/m7-assignment-and-polish.mdx).
// email set = email-bound invite (only that account can accept); email null =
// open invite link anyone signed in can accept. Acceptance creates an Enrollment.
model Invite {
  id           String    @id @default(cuid())
  token        String    @unique
  courseId     String
  course       Course    @relation(fields: [courseId], references: [id], onDelete: Cascade)
  email        String?
  invitedById  String?
  invitedBy    User?     @relation("InviteSender", fields: [invitedById], references: [id], onDelete: SetNull)
  acceptedAt   DateTime?
  acceptedById String?
  revokedAt    DateTime?
  expiresAt    DateTime?
  createdAt    DateTime  @default(now())
  @@index([courseId])
  @@index([email])
}

// Denormalized per-block aggregates, maintained in the same transaction as each
// Response write (src/server/progress/compute.ts) so reports never aggregate
// over Response on the read path — see /AGENTS.md invariant 7.
model BlockStats {
  blockId         String    @id
  block           Block     @relation(fields: [blockId], references: [id], onDelete: Cascade)
  attempts        Int       @default(0)
  correctCount    Int       @default(0)
  pendingReviews  Int       @default(0)
  lastSubmittedAt DateTime?
}
```

Auth.js v5 additionally requires its standard `Account`, `Session`, and `VerificationToken` models
via `@auth/prisma-adapter` — add those in M0 alongside `User` (see
[`phases/m0-foundation.mdx`](phases/m0-foundation.mdx)); they're omitted above since they're
boilerplate from the adapter's own docs, not domain-specific.

## Response payload shapes

```ts
type ResponsePayload =
  | { type: 'QUIZ';          selected: string[] }
  | { type: 'TEST';          answers: Record<string, string[]>; startedAt: string }
  | { type: 'MUST_READ';     acknowledged: true; openedAt?: string; note?: string }
  | { type: 'OPEN_QUESTION'; text: string; wordCount: number }
  | { type: 'CODE';          files: Record<string,string>; stdout?: string;
                             checksPassed?: number; checksTotal?: number; ranAt: string }
```

## Answer submission flow

1. Client `POST`s to `/api/blocks/[blockId]/respond` with a typed payload.
2. Server: authenticate → verify enrollment → verify block belongs to an accessible chapter →
   verify gating allows it (see [`04-progress-and-gating.mdx`](04-progress-and-gating.mdx)) →
   verify attempt limits.
3. Look up `Block.config` (server-only) and run the type's grader from the registry (see
   [`03-blocks-registry.mdx`](03-blocks-registry.mdx)).
4. In **one transaction**: insert `Response`, upsert `ChapterProgress`, recompute `Enrollment`
   rollup, enqueue a review row if the type needs one.
5. Return the sanitized result: score, correctness, explanation (if the block reveals it), and
   updated progress.

**Never** send `Block.config` to the client. The sanitizer in `mdx/sanitize.ts` produces the client
projection and must be covered by a test that fails if a `correct`, `rubric`, `sampleAnswer`, or
`hiddenTests` key ever reaches the client payload — see [`09-security.mdx`](09-security.mdx).
